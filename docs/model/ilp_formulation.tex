\section{Parameters}
\begin{tabular}{ll}
\toprule
Symbol & Meaning \\
\midrule
$l_p \in \mathbb{Z}_{\ge 0},\ u_p \in \mathbb{Z}_{\ge 1},\ l_p \le u_p$ & min.\ and max.\ headcount of project $p$ \\
$\alpha_{s,t} \in \{0,1,\dots,5\}$ & proficiency of $s$ in skill $t$ ($0$ = not held) (R7) \\

\bottomrule

\medskip\noindent
Alternatives are indexed $k = 1,\dots,K$; $j$ denotes an earlier alternative.

\paragraph{Preference rank (R14).} An unranked project receives the penalty rank $m+1$:
\begin{equation}
r_{s,p} =
\begin{cases}
\text{stated rank of } p \text{ by } s, & \text{if } s \text{ ranked } p,\\
m + 1, & \text{otherwise.}
\end{cases}
\end{equation}

\paragraph{Normalised preference score.} $\phi_{s,p} \in [0,1]$, equal to $1$ for a first choice and $0$ for an unranked project:
\begin{equation}
\phi_{s,p} = \frac{m + 1 - r_{s,p}}{m}.
\end{equation}

\paragraph{Skill fit (R4).} Computed over non-mandatory requirements only:
\begin{equation}
\sigma_{s,p} =
\begin{cases}
\dfrac{1}{|T_p^{\soft}|} \displaystyle\sum_{t \in T_p^{\soft}} \min\!\left(1, \frac{\alpha_{s,t}}{\beta_{p,t}}\right), & T_p^{\soft} \neq \emptyset,\\[2ex]
1, & T_p^{\soft} = \emptyset.
\end{cases}
\end{equation}
Note that $r_{s,p}$, $\phi_{s,p}$ and $\sigma_{s,p}$ are all \emph{data}: they are computed before solving, so every objective
below is linear in the decision variables.

\section{Decision variables}
\begin{alignat}{3}
& x_{s,p} \in \{0,1\} \qquad && \forall s \in S,\ p \in P \qquad && \text{1 iff student } s \text{ is assigned to project } p \\
& y_p \in \{0,1\} && \forall p \in P^{E} && \text{1 iff project } p \text{ is used (R13)} \\
& \theta \in \mathbb{R}_{\ge 0} && && \text{worst assigned rank (\textsc{Balanced} only)}
\end{alignat}

\section{Hard constraints (all strategies)}
These are never relaxed. If they cannot all hold, the run is reported as infeasible with diagnostics
(R2, R10).
\begin{alignat}{3}
& \sum_{p \in P} x_{s,p} = 1 && \forall s \in S && \text{(each student exactly once, R1)} \label{c:one}\\
& l_p \le \sum_{s \in S} x_{s,p} \le u_p && \forall p \in P && \text{(headcount bounds, R1)} \label{c:cap}\\
& x_{s,p} = 1 && \forall (s,p) \in F && \text{(fixed assignments, R3)} \label{c:fix}\\
& \sum_{s \in Q_{p,t}} x_{s,p} \ge y_p \qquad && \forall p \in P^{E},\ t \in E_p \qquad && \text{(mandatory skill covered, R13)} \label{c:mand}\\
& \sum_{s \in S} x_{s,p} \le u_p \, y_p && \forall p \in P^{E} && \text{(non-empty} \Rightarrow \text{used, R13)} \label{c:used}
\end{alignat}
Constraint \eqref{c:used} forces $y_p = 1$ whenever project $p$ receives a student, which activates \eqref{c:mand}. A project
with $l_p = 0$ may therefore stay empty ($y_p = 0$) without violating its mandatory requirements; if
$l_p \ge 1$, \eqref{c:cap} makes it non-empty and the requirement must be met.

\paragraph{No-good cuts for distinct alternatives (R5).} Let $\Gamma^{(j)} = \{(s,p) : x^{(j)}_{s,p} = 1\}$ be the pairs of the
$j$-th alternative found. Before solving for alternative $k$ ($k = 2,\dots,K$), add for every earlier alternative
\begin{equation}
\sum_{(s,p) \in \Gamma^{(j)}} x_{s,p} \le n - 1 \qquad j = 1,\dots,k-1. \label{c:cut}
\end{equation}
Since $|\Gamma^{(j)}| = n$, at least one student must move. If the model becomes infeasible, no further
distinct alternative exists; the solutions found so far are returned and their number is reported in
\texttt{diagnostics["distinct\_alternatives"]}.

\section{Objective functions (strategies)}
\paragraph{Order of strategies (R15).} Alternative 1 uses the strategy selected by the user; alternative $k$
uses the next strategy in $\Omega$ (cyclically) together with the cuts \eqref{c:cut}.

\paragraph{Ties (R16).} Each strategy has a secondary term with a small weight; any remaining exact tie
is resolved by the fixed sorted model order and a single-threaded HiGHS run.

\subsection{Strategy 1 --- Preference: minimise total rank}
\begin{equation}
\boxed{
\begin{aligned}
\min_{x,y}\quad & Z_{\text{pref}} = \sum_{s \in S}\sum_{p \in P} r_{s,p}\,x_{s,p} \;-\; \varepsilon_{\text{pref}} \sum_{s \in S}\sum_{p \in P} \sigma_{s,p}\,x_{s,p},
\qquad \varepsilon_{\text{pref}} = \frac{1}{n+1}\\
\text{s.t.}\quad & \eqref{c:one}\text{--}\eqref{c:used},\ \text{cuts } \eqref{c:cut}
\end{aligned}}
\end{equation}
The rank sum is an integer and the skill term is bounded by $n\,\varepsilon_{\text{pref}} < 1$, so the secondary term
only breaks ties and can never worsen the rank sum. Minimising the rank sum is equivalent to
minimising the mean rank $\bar r$.

\subsection{Strategy 2 --- SkillFit: maximise total skill fit}
\begin{equation}
\boxed{
\begin{aligned}
\max_{x,y}\quad & Z_{\text{skill}} = \sum_{s \in S}\sum_{p \in P} \sigma_{s,p}\,x_{s,p} \;+\; \varepsilon_{\text{skill}} \sum_{s \in S}\sum_{p \in P} \phi_{s,p}\,x_{s,p},
\qquad \varepsilon_{\text{skill}} = \frac{10^{-3}}{n}\\
\text{s.t.}\quad & \eqref{c:one}\text{--}\eqref{c:used},\ \text{cuts } \eqref{c:cut}
\end{aligned}}
\end{equation}
The preference term is at most $10^{-3}$. Because skill-fit totals are not integers, this is a practical
(not exact) lexicographic guarantee; we state this as a limitation.

\subsection{Strategy 3 --- Balanced: protect the worst-off, then balance}
Solved in two stages, so no weight between fairness and efficiency has to be tuned.

\paragraph{Stage 1 (egalitarian / min--max rank):}
\begin{equation}
\boxed{
\begin{aligned}
\theta^{*} = \min_{x,y,\theta}\quad & \theta\\
\text{s.t.}\quad & \theta \ge \sum_{p \in P} r_{s,p}\,x_{s,p} \qquad \forall s \in S\\
& \eqref{c:one}\text{--}\eqref{c:used},\ \text{cuts } \eqref{c:cut}
\end{aligned}}
\end{equation}

\paragraph{Stage 2 (balance among min--max optimal solutions):}
\begin{equation}
\boxed{
\begin{aligned}
\max_{x,y}\quad & Z_{\text{bal}} = \sum_{s \in S}\sum_{p \in P} \bigl(\lambda\,\phi_{s,p} + (1-\lambda)\,\sigma_{s,p}\bigr)\,x_{s,p}, \qquad \lambda = 0.5\\
\text{s.t.}\quad & \sum_{p \in P} r_{s,p}\,x_{s,p} \le \theta^{*} \qquad \forall s \in S\\
& \eqref{c:one}\text{--}\eqref{c:used},\ \text{cuts } \eqref{c:cut}
\end{aligned}}
\end{equation}
Both $\phi_{s,p}$ and $\sigma_{s,p}$ lie in $[0,1]$, so $\lambda = 0.5$ weighs them equally. Stage 1 guarantees that no student
receives a worse rank than strictly necessary.

\section{Complete model at a glance (Strategy 1 shown)}
\begin{alignat*}{2}
\min\quad & \sum_{s \in S}\sum_{p \in P} \Bigl(r_{s,p} - \tfrac{1}{n+1}\,\sigma_{s,p}\Bigr)\,x_{s,p} \\
\text{s.t.}\quad & \sum_{p \in P} x_{s,p} = 1 && \forall s \in S\\
& l_p \le \sum_{s \in S} x_{s,p} \le u_p && \forall p \in P\\
& x_{s,p} = 1 && \forall (s,p) \in F\\
& \sum_{s \in Q_{p,t}} x_{s,p} \ge y_p && \forall p \in P^{E},\ t \in E_p\\
& \sum_{s \in S} x_{s,p} \le u_p\,y_p && \forall p \in P^{E}\\
& \sum_{(s,p) \in \Gamma^{(j)}} x_{s,p} \le n - 1 \qquad && j = 1,\dots,k-1\\
& x_{s,p} \in \{0,1\},\quad y_p \in \{0,1\}
\end{alignat*}

\paragraph{Size.} $nm + |P^{E}|$ binary variables; $n + 2m + |F| + \sum_{p \in P^{E}} (|E_p| + 1) + (k-1)$ constraints (plus $n$
for each \textsc{Balanced} stage).
